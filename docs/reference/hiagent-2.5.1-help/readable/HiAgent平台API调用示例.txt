  HiAgent 平台 API 调用示例
  本文适用于 V1.4.x 及以后版本。

  调用 API 需要获取如下参数：
1. Action：接口名

1. Version：固定为 2023-08-01

1. 请求地址

   1. TopHost：top 服务地址（根据部署情况用户自行配置 top 服务对外访问地址）

      Python
      # kubectl get svc -n vke-system       volc-top -owide

      AGE    SELECTOR
      volc-top   NodePort   xxx   <none>
      8000:30040/TCP,56790:30041/TCP   268d    app=volc-
      top,release=volc-top



      TopHost 地址解释：hiagent 环境会部署 volc-top 服务，端口地址为 30040，外
      部用户一般不能直接访问，需要环境对应的运维人员暴露这个地址用于 API 调
      用

1. 主账号

   1. AccountID：默认值一般为 1000000000，可以在         运营端->租户管理获取




1. 签名（请求过网关需要签名，这里 sign 参数会生成签名附带在请求头 Authorization 中）
1. AccessKeyID、SecretAccessKey：通过   点击头像 -> 账号中心 -> 密钥管理 获取




1. Region：区域（没特殊要求就用这个:cn-north-1）

1. Service：服务名，取值如下

    1.aigw: ai 网关

    1.app: 智能体

    1.eva: 评测

    1.iam: iam

    1.observe: 观测

    1.up: 文件上传

1. Version：对应服务接口版本，取值如下

    1.aigw version: 2023-08-01

    1.app version: 2023-08-01

    1.eva version: 2025-02-01

    1.iam version: 2024-12-25

    1.observe version: 2025-05-01

    1.up version: 2022-01-01

class SignerV4(object):
    @staticmethod
    def sign(
        path,
        method,
        headers,
        body,
        query,
        ak,
        sk,
        region,
        service,
        use_sm: bool = False,
    ):
        if path == "":
            path = "/"
        if method != "GET" and not (
            "Content-Type" in headers or "content-type" in headers
        ):
            headers["Content-Type"] = "application/x-www-form-
urlencoded; charset=utf-8"
        format_date =
datetime.datetime.utcnow().strftime("%Y%m%dT%H%M%SZ")
        headers["X-Date"] = format_date

        body_hash = SignerV4.hashSum(body.encode("utf-8"),
use_sm).hex()
        headers["X-Content-Sha256"] = body_hash

       signed_headers = dict()
       for key in headers:
           if key in [
               "Content-Type",
               "content-type",
               "Content-Md5",
               "Host",
           ] or key.startswith("X-"):
               signed_headers[key.lower()] = headers[key]

       if "host" in signed_headers:
           v = signed_headers["host"]
           if v.find(":") != -1:
               split = v.split(":")
               port = split[1]
               if str(port) == "80" or str(port) == "443":
                   signed_headers["host"] = split[0]

       signed_str = ""
       for key in sorted(signed_headers.keys()):
           signed_str += key + ":" + signed_headers[key] + "\n"

        signed_headers_string =
";".join(sorted(signed_headers.keys()))

       canonical_request = "\n".join(
           [
               method,
               path,
               SignerV4.canonical_query(dict(query)),
               signed_str,
               signed_headers_string,
               body_hash,
           ]
       )
       credential_scope = "/".join([format_date[:8], region,
service, "request"])
        signing_str = "\n".join(
            [
                "HMAC-SHA256",
                format_date,
                credential_scope,

SignerV4.hashSum(canonical_request.encode("utf-8"), use_sm).hex(),
            ]
        )
        signing_key = SignerV4.get_signing_secret_key_v4(
            sk, format_date[:8], region, service, use_sm
        )

       signature = SignerV4.hmac_sum(
           signing_key, signing_str.encode("utf-8"), use_sm
       ).hex()

       credential = ak + "/" + credential_scope
       headers["Authorization"] = (
           "HMAC-SHA256"
           + " Credential="
           + credential
           + ", SignedHeaders="
           + signed_headers_string
           + ", Signature="
           + signature
       )
       return

    @staticmethod
    def canonical_query(query):
        res = []
        for key in query:
            value = str(query[key])
            res.append((quote(key, safe="-_.~"), quote(value,
safe="-_.~")))
        sorted_key_vals = []
        for key, value in sorted(res):
            sorted_key_vals.append("%s=%s" % (key, value))
        return "&".join(sorted_key_vals)

   @staticmethod
   def get_signing_secret_key_v4(sk, date, region, service,
use_sm: bool = False):
        kdate = SignerV4.hmac_sum(sk.encode("utf-8"),
date.encode("utf-8"), use_sm)
        kregion = SignerV4.hmac_sum(kdate, region.encode("utf-8"),
use_sm)
        kservice = SignerV4.hmac_sum(kregion,
service.encode("utf-8"), use_sm)
        return SignerV4.hmac_sum(kservice,
"request".encode("utf-8"), use_sm)

   @staticmethod
   def hmac_sum(key: bytes, msg: bytes, use_sm: bool = False):
       hasher = hashes.SHA256()
       if use_sm:
           hasher = hashes.SM3()
       h = hmac.HMAC(key, hasher)
       h.update(msg)
       return h.finalize()

   @staticmethod
   def hashSum(msg: bytes, use_sm: bool = False):

          * Creates a canonical query string from a map of query
parameters.
          */
         public static String canonicalQuery(Map<String, String>
query) {
             if (query == null || query.isEmpty()) {
                 return "";
             }
            return query.entrySet().stream()
                    .sorted(Map.Entry.comparingByKey())
                    .map(e -> urlEncode(e.getKey()) + "=" +
urlEncode(e.getValue()))
                    .collect(Collectors.joining("&"));
        }

        private static String urlEncode(String value) {
            try {
                return URLEncoder.encode(value,
StandardCharsets.UTF_8.name())
                        .replace("+", "%20")
                        .replace("*", "%2A")
                        .replace("%7E", "~");
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }

        /**
          * Derives the signing key for Signature Version 4.
          */
        private static byte[] getSigningSecretKeyV4(String sk,
String date, String region, String service, boolean useSm) throws
InvalidKeyException, NoSuchAlgorithmException {
             byte[] kDate =
hmacSum(sk.getBytes(StandardCharsets.UTF_8),
date.getBytes(StandardCharsets.UTF_8), useSm);
             byte[] kRegion = hmacSum(kDate,
region.getBytes(StandardCharsets.UTF_8), useSm);
             byte[] kService = hmacSum(kRegion,
service.getBytes(StandardCharsets.UTF_8), useSm);
             return hmacSum(kService,
"request".getBytes(StandardCharsets.UTF_8), useSm);
        }

        /**
         * Calculates HMAC signature.
         */
        private static byte[] hmacSum(byte[] key, byte[] msg,
boolean useSm) throws NoSuchAlgorithmException,
InvalidKeyException {
            String alg = useSm ? "HmacSM3" : "HmacSHA256";
            Mac mac = Mac.getInstance(alg);
            mac.init(new SecretKeySpec(key, alg));
            return mac.doFinal(msg);
        }

        /**
          * Calculates a hash (SHA-256 or SM3).
          */
        private static byte[] hashSum(byte[] msg, boolean useSm)
throws NoSuchAlgorithmException {
             String alg = useSm ? "SM3" : "SHA-256";
             MessageDigest digest = MessageDigest.getInstance(alg);
             return digest.digest(msg);
        }

        private static final char[] HEX_ARRAY =
"0123456789abcdef".toCharArray();

        /**
          * Converts a byte array to a hexadecimal string.
          */
        public static String bytesToHex(byte[] bytes) {
             char[] hexChars = new char[bytes.length * 2];
             for (int j = 0; j < bytes.length; j++) {
                 int v = bytes[j] & 0xFF;
                 hexChars[j * 2] = HEX_ARRAY[v >>> 4];
                 hexChars[j * 2 + 1] = HEX_ARRAY[v & 0x0F];
             }
             return new String(hexChars);
        }
    }

func hashedSimpleCanonicalRequestV4(request *http.Request, query
url.Values, meta *metadata) string {
    payloadHash := hashSum([]byte(""))

    if request.URL.Path == "" {
       request.URL.Path = "/"
    }

    canonicalRequest := concat("\n", request.Method,
normuri(request.URL.Path), normquery(query), "\n",
meta.signedHeaders, payloadHash)

    return hashSum([]byte(canonicalRequest))
}

func hashedCanonicalRequestV4(request *http.Request, meta
*metadata) string {
    payload := readAndReplaceBody(request)
    payloadHash := hashSum(payload)
    request.Header.Set("X-Content-Sha256", payloadHash)

    request.Header.Set("Host", request.Host)
    var sortedHeaderKeys []string
    for key := range request.Header {
       switch key {
       case "Content-Type", "Content-Md5", "Host":
       default:
          if !strings.HasPrefix(key, "X-") {
             continue
          }
       }
       sortedHeaderKeys = append(sortedHeaderKeys,
strings.ToLower(key))
    }
    sort.Strings(sortedHeaderKeys)

    var headersToSign string
    for _, key := range sortedHeaderKeys {
       value := strings.TrimSpace(request.Header.Get(key))
       if key == "host" {
          if strings.Contains(value, ":") {
             split := strings.Split(value, ":")
             port := split[1]
             if port == "80" || port == "443" {
                value = split[0]
             }
          }
       }
       headersToSign += key + ":" + value + "\n"
    }
    meta.signedHeaders = concat(";", sortedHeaderKeys...)
    canonicalRequest := concat("\n", request.Method,
normuri(request.URL.Path), normquery(request.URL.Query()),
headersToSign, meta.signedHeaders, payloadHash)

    return hashSum([]byte(canonicalRequest))
}

func stringToSignV4(request *http.Request, hashedCanonReq string,

    requestTs := request.Header.Get("X-Date")

    meta.algorithm = "HMAC-SHA256"
    meta.date = tsDateV4(requestTs)
    meta.credentialScope = concat("/", meta.date, meta.region,
meta.service, "request")
    return concat("\n", meta.algorithm, requestTs,
meta.credentialScope, hashedCanonReq)
}

func signatureV4(signingKey []byte, stringToSign string) string {
    return hex.EncodeToString(hmacSign(signingKey, stringToSign))
}

func prepareRequestV4(request *http.Request) *http.Request {
    necessaryDefaults := map[string]string{
       "Content-Type": "application/x-www-form-urlencoded;
charset=utf-8",
       "X-Date":       timestampV4(),
    }

    for header, value := range necessaryDefaults {
       if request.Header.Get(header) == "" {
          request.Header.Set(header, value)
       }
    }

    if request.URL.Path == "" {
       request.URL.Path += "/"
    }

    return request
}

}

func readAndReplaceBody(request *http.Request) []byte {
    if request.Body == nil {
       return []byte{}
    }
    payload, _ := io.ReadAll(request.Body)
    request.Body = io.NopCloser(bytes.NewReader(payload))
    return payload
}

func normuri(uri string) string {
    parts := strings.Split(uri, "/")
    for i := range parts {
       parts[i] = encodePathFrag(parts[i])
    }
    return strings.Join(parts, "/")
}

func encodePathFrag(s string) string {
    hexCount := 0
    for i := 0; i < len(s); i++ {
       c := s[i]
       if shouldEscape(c) {
           hexCount++
       }
    }
    t := make([]byte, len(s)+2*hexCount)
    j := 0
    for i := 0; i < len(s); i++ {
       c := s[i]
       if shouldEscape(c) {
           t[j] = '%'
           t[j+1] = "0123456789ABCDEF"[c>>4]
           t[j+2] = "0123456789ABCDEF"[c&15]
           j += 3
       } else {
           t[j] = c
           j++
       }
    }
    return string(t)
}

func shouldEscape(c byte) bool {
    if 'a' <= c && c <= 'z' || 'A' <= c && c <= 'Z' {
       return false
    }
    if '0' <= c && c <= '9' {
       return false
    }
    if c == '-' || c == '_' || c == '.' || c == '~' {
       return false
    }
    return true
}
func normquery(v url.Values) string {

    return strings.ReplaceAll(queryString, "+", "%20")

func NewHmacSM3() MacHasher {
    return &hmacCipher{
       macFactor: hmac.New,
       hashFactor: sm3.New,
    }
}

type hmacCipher struct {
    macFactor func(h func() hash.Hash, key []byte) hash.Hash
    hashFactor func() hash.Hash
}

func (a *hmacCipher) Sign(message []byte, key string) ([]byte,
error) {
    h := a.macFactor(a.hashFactor, []byte(key))
    h.Write([]byte(message))
    return h.Sum(nil), nil
}
